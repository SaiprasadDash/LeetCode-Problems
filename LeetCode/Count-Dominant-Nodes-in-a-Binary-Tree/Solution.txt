1/**
2 * Definition for a binary tree node.
3 * public class TreeNode {
4 *     int val;
5 *     TreeNode left;
6 *     TreeNode right;
7 *     TreeNode() {}
8 *     TreeNode(int val) { this.val = val; }
9 *     TreeNode(int val, TreeNode left, TreeNode right) {
10 *         this.val = val;
11 *         this.left = left;
12 *         this.right = right;
13 *     }
14 * }
15 */
16class Solution {
17    int c = 0;
18    public int countDominantNodes(TreeNode root) {
19        dfs(root);
20        return c;
21        
22    }
23    public int dfs(TreeNode root){
24        if (root == null) return 0;
25        
26        int l = dfs(root.left);
27        int r = dfs(root.right);
28
29        int max = Math.max(root.val, Math.max(l, r));
30        if(root.val == max) c++;
31
32        return max;
33    }
34}
